// REFERENCE ONLY, NOT PART OF THE BUILD.  UNTESTED, never simulated or wired in.
// An earlier draft of the OLED plot, kept as an optional starting point for M1.
// The history RAM and rotl() read now live in m2_astar/ast_perf_hist.v, so a
// renderer should take its levels from that module instead (CONTRACT 8a).
`timescale 1ns / 1ps
`include "arena_defs.vh"
//////////////////////////////////////////////////////////////////////////////
// ast_oled_graph : the progress plot on the OLED results view (CONTRACT 8a)
//
// Holds the sampler's history and turns pixel_index into a plot pixel.  The
// RAMs are written on clk (100 MHz) and read combinationally from the 6.25 MHz
// pixel_index; a read during a write can at worst mis-colour one pixel for one
// frame.  Two copies give the current and the previous column in one pixel.
//
//   x = 0     y axis            sample i is drawn at x = i + 1
//   y = 63    x axis            level L  is drawn at y = 62 - L
//
// A curve runs from its previous level to its current one in each column, so
// it stays continuous.  It stops at the engine's finish, where a dotted line
// runs down to the x axis.  Colours match VGA; A* is drawn on top.
//////////////////////////////////////////////////////////////////////////////
module ast_oled_graph #(
    parameter AW        = `ARENA_PG_ADDR_W,
    parameter RW        = `ARENA_PG_ROT_W,
    parameter LW        = `ARENA_PG_LEVEL_W,
    parameter LEVEL_MAX = `ARENA_PG_LEVEL_MAX,
    parameter OLED_W    = `ARENA_OLED_W,
    parameter OLED_H    = `ARENA_OLED_H
) (
    input  wire              clk,
    input  wire              we,
    input  wire [AW-1:0]     waddr,
    input  wire [4*LW-1:0]   wdata,
    input  wire [AW-1:0]     count,
    input  wire [RW-1:0]     rot,
    input  wire [4*AW-1:0]   end_idx,
    input  wire [3:0]        end_valid,
    input  wire [3:0]        mask,

    input  wire [12:0]       pixel_index,
    output wire              on,           // this pixel belongs to the plot area
    output reg  [15:0]       pixel_data
);
    localparam AXIS_Y = OLED_H - 1;
    localparam TOP_Y  = AXIS_Y - 1 - LEVEL_MAX;
    localparam [15:0] BLACK = 16'h0000, C_AXIS = 16'h7BEF;
    localparam [15:0] C_BFS = 16'h07FF, C_DIJ = 16'hFFE0, C_AST = 16'h07E0, C_DFS = 16'hFB35;

    wire [6:0] px = pixel_index % OLED_W;
    wire [5:0] py = pixel_index / OLED_W;
    assign on = (py >= TOP_Y);

    // ---- history -----------------------------------------------------------
    wire [AW-1:0] idx      = px - 1'b1;                     // sample in this column
    wire [AW-1:0] idx_prev = (idx == 0) ? {AW{1'b0}} : idx - 1'b1;
    wire [2*AW-1:0] cur_sh  = {idx, idx} >> (AW - rot);     // rotl(idx, rot)
    wire [2*AW-1:0] prev_sh = {idx_prev, idx_prev} >> (AW - rot);
    wire [4*LW-1:0] cur_w, prev_w;
    grid_mem #(.ADDR_W(AW), .DATA_W(4*LW)) u_cur (
        .clk(clk), .we(we), .waddr(waddr), .wdata(wdata),
        .raddr(cur_sh[AW-1:0]), .rdata(cur_w)
    );
    grid_mem #(.ADDR_W(AW), .DATA_W(4*LW)) u_prev (
        .clk(clk), .we(we), .waddr(waddr), .wdata(wdata),
        .raddr(prev_sh[AW-1:0]), .rdata(prev_w)
    );

    // ---- curves ------------------------------------------------------------
    wire          in_plot = (px != 0) && on && (py != AXIS_Y) && (idx < count);
    wire [LW-1:0] lv      = AXIS_Y - 1 - py;                // level of this row
    reg  [3:0]    curve, fin;
    reg  [LW-1:0] c_lv, p_lv;
    reg  [AW-1:0] e_i;
    integer e;
    always @* begin
        for (e = 0; e < 4; e = e + 1) begin
            c_lv = cur_w[e*LW +: LW];
            p_lv = prev_w[e*LW +: LW];
            e_i  = end_idx[e*AW +: AW];
            curve[e] = in_plot && mask[e] && (!end_valid[e] || idx <= e_i) &&
                       (lv >= p_lv) && (lv <= c_lv);
            fin[e]   = in_plot && mask[e] && end_valid[e] && (idx == e_i) &&
                       (lv < c_lv) && py[0];
        end
    end

    always @* begin
        if      (!on)                      pixel_data = BLACK;
        else if (curve[`ARENA_ENG_AST])    pixel_data = C_AST;
        else if (curve[`ARENA_ENG_DIJ])    pixel_data = C_DIJ;
        else if (curve[`ARENA_ENG_BFS])    pixel_data = C_BFS;
        else if (curve[`ARENA_ENG_DFS])    pixel_data = C_DFS;
        else if (fin[`ARENA_ENG_AST])      pixel_data = C_AST;
        else if (fin[`ARENA_ENG_DIJ])      pixel_data = C_DIJ;
        else if (fin[`ARENA_ENG_BFS])      pixel_data = C_BFS;
        else if (fin[`ARENA_ENG_DFS])      pixel_data = C_DFS;
        else if (px == 0 || py == AXIS_Y)  pixel_data = C_AXIS;
        else                               pixel_data = BLACK;
    end
endmodule
